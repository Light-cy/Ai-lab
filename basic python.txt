# ===== TASK 14: Binary search recursive =====
def task14():
    def bin_search(arr, low, high, target):
        if low > high:
            return -1
        mid = (low + high) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            return bin_search(arr, mid + 1, high, target)
        else:
            return bin_search(arr, low, mid - 1, target)
    numbers = [1, 3, 5, 7]
    target = int(input("Target: "))
    print("Index:", bin_search(numbers, 0, len(numbers) - 1, target))

# ===== TASK 15: Quick sort =====
def task15():
    def quick(arr):
        if len(arr) <= 1:
            return arr
        pivot = arr[0]
        left = []
        right = []
        for x in arr[1:]:
            if x < pivot:
                left.append(x)
            else:
                right.append(x)
        return quick(left) + [pivot] + quick(right)
    numbers = [5, 3, 8, 4]
    print("Sorted:", quick(numbers))

# ===== TASK 17: Vigenere cipher encrypt =====
def task17():
    text = input("Text: ").upper()
    key = input("Key: ").upper()
    encrypted = ""
    k = 0
    for char in text:
        if char >= 'A' and char <= 'Z':
            shift = ord(key[k % len(key)]) - ord('A')
            encrypted += chr((ord(char) - ord('A') + shift) % 26 + ord('A'))
            k += 1
        else:
            encrypted += char
    print("Encrypted:", encrypted)

# ===== TASK 39: Merge two sorted arrays =====
def task39():
    arr1 = [1,3,5]
    arr2 = [2,4,6]
    merged = []
    i = 0
    j = 0
    while i < len(arr1) and j < len(arr2):
        if arr1[i] < arr2[j]:
            merged.append(arr1[i])
            i += 1
        else:
            merged.append(arr2[j])
            j += 1
    while i < len(arr1):
        merged.append(arr1[i])
        i += 1
    while j < len(arr2):
        merged.append(arr2[j])
        j += 1
    print("Merged:", merged)

# ===== TASK 41: Simple bubble sort descending =====
def task41():
    numbers = [5,3,8,4]
    n = len(numbers)
    for i in range(n):
        for j in range(0, n - i - 1):
            if numbers[j] < numbers[j + 1]:
                temp = numbers[j]
                numbers[j] = numbers[j + 1]
                numbers[j + 1] = temp
    print("Desc sorted:", numbers)
